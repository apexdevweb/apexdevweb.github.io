// Boule de feu WebGL intégrée sur le curseur de défilement horizontal de la règle.
// Usage : <script src="fireball-cursor.js"></script>
// Cible : <div class="cursor-index"></div> à l'intérieur de .ruler

// ---------- Réglages ----------
const SPEED_SENSITIVITY = 2; //12 ||Plus le chiffre est bas, plus la flamme s'étire vite lors du mouvement
const SMOOTH_ENERGY = 0.1; //0.18 ||Vitesse de réaction et de retombée de la flamme (0.1 = traînant, 0.9 = instantané)
const BLEND = "screen"; // Mode de fusion ("screen" = fusion lumineuse avec les lignes en dessous, "normal" = opaque)

(() => {
  // Sélection du curseur dans la structure de votre règle
  const targetDiv =
    document.querySelector(".main___ruler--ctnr .cursor-index") ||
    document.querySelector(".cursor-index");
  if (!targetDiv) {
    console.error("Élément .cursor-index introuvable dans la structure HTML.");
    return;
  }

  // Création du canvas WebGL qui va générer le feu
  const canvas = document.createElement("canvas");
  const gl = canvas.getContext("webgl", {
    alpha: true,
    premultipliedAlpha: true,
    antialias: false,
  });
  if (!gl) return;

  // ---------- Shaders ----------
  const VERT = `
    attribute vec2 aPos;
    void main(){ gl_Position = vec4(aPos, 0.0, 1.0); }
  `;

  const FRAG = `
    precision highp float;
    uniform vec2  uRes;
    uniform float uTime;
    uniform float uEnergy; 
    uniform vec2  uDir;    

    float hash(vec2 p){
      p = fract(p * vec2(123.34, 456.21));
      p += dot(p, p + 45.32);
      return fract(p.x * p.y);
    }
    float noise(vec2 p){
      vec2 i = floor(p), f = fract(p);
      f = f * f * (3.0 - 2.0 * f);
      float a = hash(i);
      float b = hash(i + vec2(1.0, 0.0));
      float c = hash(i + vec2(0.0, 1.0));
      float d = hash(i + vec2(1.0, 1.0));
      return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
    }
    float fbm(vec2 p){
      float v = 0.0, a = 0.5;
      for (int i = 0; i < 5; i++){
        v += a * noise(p);
        p = p * 2.02 + vec2(17.3, 9.1);
        a *= 0.5;
      }
      return v;
    }

    void main(){
      // Normalisation basée sur l'axe Y pour préserver les proportions sphériques du cœur de la boule
      vec2 p = (gl_FragCoord.xy - 0.5 * uRes) / (0.5 * uRes.y); 
      vec2 d = normalize(uDir);
      float along = dot(p, d);
      float perp  = dot(p, vec2(-d.y, d.x));
      float e = uEnergy;

      // Étirement horizontal de la flamme proportionnel à la vitesse
      float stretch = 1.0 + e * 2.2;
      float a = along > 0.0 ? along / stretch : along * 1.1;
      float r = length(vec2(a, perp));

      // Vitesse d'écoulement du fluide enflammé
      vec2 flow = p * 2.6 - d * uTime * (1.3 + e * 3.2);
      float n = fbm(flow + fbm(flow * 1.5 + uTime * 0.2) * 0.85);

      float tailMask = smoothstep(-0.4, 0.6, along);
      float radius = 0.35 + 0.05 * e;
      float edge = radius + (n - 0.45) * (0.18 + 0.35 * e) * (0.35 + 0.65 * tailMask);
      float body = 1.0 - smoothstep(edge - 0.30, edge, r);

      // Gradient thermique (Blanc chaud -> Jaune ardent -> Orange -> Rouge sombre)
      float heat = clamp(body * (1.2 - r / max(edge, 0.05)) + n * 0.25 * body, 0.0, 1.2);
      vec3 col = mix(vec3(0.78, 0.04, 0.0), vec3(1.0, 0.42, 0.02), smoothstep(0.12, 0.50, heat));
      col = mix(col, vec3(1.0, 0.84, 0.38), smoothstep(0.50, 0.88, heat));
      col = mix(col, vec3(1.0), smoothstep(0.88, 1.2, heat) * 0.9);

      float glow = exp(-r * 2.6) * (0.32 + 0.38 * e);
      float alpha = clamp(body * 1.05, 0.0, 1.0);

      // Masque pour adoucir le rendu aux extrémités du rectangle
      float mask = 1.0 - smoothstep(0.82, 1.0, abs(p.x / (uRes.x / uRes.y)));

      float al = clamp(alpha + glow, 0.0, 1.0) * mask;
      vec3 c = (col * alpha + vec3(1.0, 0.35, 0.05) * glow) * mask;
      c = min(c, vec3(al));
      gl_FragColor = vec4(c, al);
    }
  `;

  function compile(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    return gl.getShaderParameter(s, gl.COMPILE_STATUS) ? s : null;
  }

  const vs = compile(gl.VERTEX_SHADER, VERT);
  const fs = compile(gl.FRAGMENT_SHADER, FRAG);
  if (!vs || !fs) return;

  const prog = gl.createProgram();
  gl.attachShader(prog, vs);
  gl.attachShader(prog, fs);
  gl.linkProgram(prog);
  gl.useProgram(prog);

  const buf = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, buf);
  gl.bufferData(
    gl.ARRAY_BUFFER,
    new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]),
    gl.STATIC_DRAW
  );
  const loc = gl.getAttribLocation(prog, "aPos");
  gl.enableVertexAttribArray(loc);
  gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);

  const uRes = gl.getUniformLocation(prog, "uRes");
  const uTime = gl.getUniformLocation(prog, "uTime");
  const uEnergy = gl.getUniformLocation(prog, "uEnergy");
  const uDir = gl.getUniformLocation(prog, "uDir");

  // ---------- Application des styles CSS ----------
  [
    ["position", "absolute"],
    ["top", "55%"],
    ["left", "50%"],
    ["transform", "translate(-50%, -50%)"],
    ["width", "100%"],
    ["height", "100%"],
    ["pointer-events", "none"],
    ["background", "none"],
  ].forEach(([k, v]) => canvas.style.setProperty(k, v, "important"));

  targetDiv.style.setProperty("mix-blend-mode", BLEND, "important");
  targetDiv.appendChild(canvas);

  // ---------- Analyse des mouvements de la règle ----------
  let lastX = targetDiv.getBoundingClientRect().left;
  let targetEnergy = 0,
    currentEnergy = 0;
  let dirX = 0,
    dirY = 1;
  let canvasW = 0,
    canvasH = 0;

  function resize() {
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const rect = targetDiv.getBoundingClientRect();

    canvasW = Math.round(rect.width * dpr);
    canvasH = Math.round(rect.height * dpr);

    if (canvas.width !== canvasW || canvas.height !== canvasH) {
      canvas.width = canvasW;
      canvas.height = canvasH;
      gl.viewport(0, 0, canvasW, canvasH);
    }
  }

  // ---------- Boucle principale d'animation ----------
  const t0 = performance.now();

  function frame(now) {
    resize();

    // Calcul de l'inertie et du déplacement horizontal dans la règle
    const currentX = targetDiv.getBoundingClientRect().left;
    const dx = currentX - lastX;
    lastX = currentX;

    if (Math.abs(dx) > 0.05) {
      targetEnergy = Math.min(1, Math.abs(dx) / SPEED_SENSITIVITY);
    } else {
      targetEnergy *= 0.82; // Amortissement à l'arrêt
    }
    currentEnergy += (targetEnergy - currentEnergy) * SMOOTH_ENERGY;

    // Définition du vecteur de direction de la traîne
    let tx = 0;
    let ty = 0.25; // Au repos, la flamme s'élève doucement vers le haut

    if (currentEnergy > 0.02) {
      // Si le curseur va vers la droite (dx > 0), la traîne s'étire vers la gauche (tx = -1)
      tx = dx > 0 ? -1.0 : 1.0;
      ty = 0.05; // Réduction de la composante verticale pour écraser le feu horizontalement
    }

    // Interpolation de la direction pour éviter les sauts brusques
    const dLen = Math.hypot(tx, ty);
    dirX += (tx / dLen - dirX) * 0.22;
    dirY += (ty / dLen - dirY) * 0.22;

    // Injection des uniforms et rendu du quad
    gl.clear(gl.COLOR_BUFFER_BIT);
    gl.uniform2f(uRes, canvasW, canvasH);
    gl.uniform1f(uTime, (now - t0) * 0.001);
    gl.uniform1f(uEnergy, currentEnergy);
    gl.uniform2f(uDir, dirX, dirY);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);

    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
