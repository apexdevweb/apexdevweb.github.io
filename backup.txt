///////////////////////////////////SCRAMBLE CANVAS/////////////////////////////////////////////////////////
////////////////////////////////////////////////////////////////////////////////////////////
// Bande de pixels orange "scramble" qui traverse l'écran en diagonale au scroll
// (du coin bas gauche vers le coin haut droit).
// Dépendances : gsap + ScrollTrigger (chargés avant ce fichier).
// HTML : <canvas id="pixel-canvas"></canvas>
// CSS  : position: fixed; inset: 0; width: 100%; height: 100lvh;
//        pointer-events: none; mix-blend-mode: difference; opacity: 0;

// Tout le code est dans une IIFE : aucune variable (CHARS, CELL, draw, setup…)
// n'est exposée globalement, donc aucun conflit avec tes autres scripts.
(() => {

  gsap.registerPlugin(ScrollTrigger);
  // Évite les recalculs quand la barre d'URL mobile apparaît / disparaît
  ScrollTrigger.config({ ignoreMobileResize: true });
  
  // ---------- Réglages ----------
  const CELL = Math.min(17, Math.max(17, Math.round(innerWidth / 190)));     // taille d'une cellule en px
  const CHARS = "01ABCDEFGHJKLMNPRSTUVXYZ#*+/<>";
  const BG = "#ff5a1f";                      // orange des cellules
  const INK = "#0f1318";                     // couleur des lettres
  const BAND_HALF = 0.17;                    // demi-épaisseur de la bande (fraction de la hauteur du canvas)
  const CORE = 0.25;                         // part de la demi-bande à densité 1 (cœur plein)
  const EDGE_FALLOFF = 3.2;                  // dispersion des bords : 1 = progressif, 2 = espacé, 3+ = très épars
  const ANGLE_DEG = 25;                      // direction de la vague en degrés au-dessus de l'horizontale
                                             // 25 = bande inclinée de 25° par rapport à la verticale : elle couvre toute la hauteur de l'écran
                                             // (10 = presque verticale, 45 = diagonale franche mais qui ne couvre la hauteur qu'au milieu de sa course)
                                             // (null = adapte l'angle pour viser exactement le coin haut droit)
  const SCRAMBLE_PX = 40;                    // les lettres changent tous les N px de scroll (plus petit = plus nerveux)
  const SCRUB = 0.6;                         // lissage du scroll (secondes)
  const REVEAL_DELAY = 2;                    // apparition du canvas (remplace l'animation CSS)
  const REVEAL_DURATION = 4;
  
  const canvas = document.getElementById("pixel-canvas");
  const ctx = canvas.getContext("2d");
  
  let W, H, dpr, cols, rows;
  let dirX, dirY, total;                     // direction de la vague et distance totale à parcourir
  let atlas = null;                          // glyphes pré-rendus (perf)
  let atlasCell = 0;
  let wasDrawn = false;
  const state = { progress: 0 };             // animé par GSAP
  
  // Bruit pseudo-aléatoire stable par cellule
  function hash(x, y) {
    let h = (x * 374761393 + y * 668265263) | 0;
    h = Math.imul(h ^ (h >>> 13), 1274126177);
    return ((h ^ (h >>> 16)) >>> 0) / 4294967295;
  }
  
  // Pré-rend chaque caractère une seule fois : drawImage est bien plus léger que fillText
  function buildAtlas() {
    atlasCell = Math.round(CELL * dpr);
    atlas = document.createElement("canvas");
    atlas.width = atlasCell * CHARS.length;
    atlas.height = atlasCell;
    const a = atlas.getContext("2d");
    a.fillStyle = INK;
    a.font = `${atlasCell * 0.85}px monospace`;
    a.textAlign = "center";
    a.textBaseline = "middle";
    for (let i = 0; i < CHARS.length; i++) {
      a.fillText(CHARS[i], i * atlasCell + atlasCell / 2, atlasCell / 2 + 1);
    }
  }
  
  function setup() {
    dpr = Math.min(devicePixelRatio || 1, 2);
    W = canvas.clientWidth;
    H = canvas.clientHeight;
    canvas.width = Math.round(W * dpr);
    canvas.height = Math.round(H * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    cols = Math.ceil(W / CELL);
    rows = Math.ceil(H / CELL);
  
    // Direction de déplacement de la vague (vecteur unitaire vers le haut droit)
    let ang = ANGLE_DEG == null ? Math.atan2(H, W) : (ANGLE_DEG * Math.PI) / 180;
    ang = Math.min(Math.max(ang, 0.09), Math.PI / 2 - 0.09); // évite les divisions par ~0
    dirX = Math.cos(ang);
    dirY = Math.sin(ang);
    total = W * dirX + H * dirY;             // distance du coin bas gauche au coin haut droit, le long de la direction
  
    buildAtlas();
    wasDrawn = false;
    draw();
  }
  
  function draw() {
    const p = state.progress;
  
    ctx.clearRect(0, 0, W, H);
  
    const halfPx = BAND_HALF * H;
    // p = 0 : bande collée au coin bas gauche / p = 1 : bande collée au coin haut droit
    const center = p * total;
    // Les lettres dépendent de la position de scroll, pas du temps : elles se figent quand on s'arrête
    const tick = Math.floor((p * ScrollTrigger.maxScroll(window)) / SCRAMBLE_PX);
  
    for (let y = 0; y < rows; y++) {
      // Distance de la ligne au bord bas, projetée sur la direction de la vague
      const base = (H - (y + 0.5) * CELL) * dirY;
  
      // Plage de colonnes réellement touchées par la bande sur cette ligne
      const xMin = Math.max(0, Math.floor((center - halfPx - base) / dirX / CELL));
      const xMax = Math.min(cols - 1, Math.ceil((center + halfPx - base) / dirX / CELL));
  
      for (let x = xMin; x <= xMax; x++) {
        const s = (x + 0.5) * CELL * dirX + base;       // position de la cellule le long de la direction
        const dist = Math.abs(s - center) / halfPx;     // 0 au centre, 1 au bord
        if (dist >= 1) continue;
  
        const t = Math.min(1, Math.max(0, (dist - CORE) / (1 - CORE)));
        const density = Math.pow(1 - t, EDGE_FALLOFF);  // plein au cœur, de plus en plus épars vers les bords
        if (hash(x, y) > density) continue;             // cellule absente
  
        ctx.fillStyle = BG;
        ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
  
        const i = Math.floor(hash(x * 31 + tick, y * 17 + tick) * CHARS.length);
        ctx.drawImage(atlas, i * atlasCell, 0, atlasCell, atlasCell, x * CELL, y * CELL, CELL, CELL);
      }
    }
  }
  
  // ---------- Animation liée au scroll ----------
  gsap.to(state, {
    progress: 1,
    ease: "none",
    onUpdate: draw,           // redessine uniquement quand le scroll fait bouger progress
    scrollTrigger: {
      trigger: document.documentElement,
      start: "top top",
      end: "bottom bottom",   // toute la page. Pour une section : trigger: ".ma-section", start: "top bottom", end: "bottom top"
      scrub: SCRUB,
    },
  });
  
  // Apparition du canvas (plus fiable que l'animation CSS sur Safari iOS)
  gsap.to(canvas, { opacity: 1, duration: REVEAL_DURATION, delay: REVEAL_DELAY, ease: "power1.inOut" });
  
  // Ne relance setup() que si la largeur change (ignore la barre d'URL mobile)
  let lastW = innerWidth;
  addEventListener("resize", () => {
    if (innerWidth === lastW) return;
    lastW = innerWidth;
    setup();
  });
  
  setup();
  
  })();
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  ///////////////////////////////////////////////////////////////////////////////////////////////////////////////
  // ---------- Réglages ----------
const CELL = 34; // taille d'un "pixel" en px
const BASE = 0.15; // hauteur de la bande au repos (fraction de la hauteur du canvas)
const BOOST = 0.5; // hauteur max de la bande pendant le scroll (1 = toute la hauteur du canvas)
const SENSITIVITY = 80; // vitesse de scroll (px/frame) pour atteindre l'effet maximum
const SMOOTH = 0.05; // fluidité de la montée / descente de l'effet (0.05 = lent, 0.3 = nerveux)
const PIXEL_COLOR = "#0f1318";

const canvas = document.getElementById("pixel-canvas");
const ctx = canvas.getContext("2d");

// Le canvas reste à sa place dans le HTML (il n'est plus déplacé en bas du body).
// Il est collé en bas de l'écran, fond transparent, sans bloquer les clics.
// Note : position: fixed est calculé par rapport à la fenêtre tant qu'aucun parent
let W, H, dpr, cols, rows;
let lastY = scrollY;
let energy = 0; // 0 = repos, 1 = scroll rapide
let target = 0;
let running = false;

// Bruit pseudo-aléatoire stable par cellule (x, y)
function hash(x, y) {
  let h = (x * 374761393 + y * 668265263) | 0;
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  return ((h ^ (h >>> 16)) >>> 0) / 4294967295;
}

function setup() {
  dpr = Math.min(devicePixelRatio || 1, 2);
  W = canvas.clientWidth;
  H = canvas.clientHeight;
  canvas.width = W * dpr;
  canvas.height = H * dpr;
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  cols = Math.ceil(W / CELL);
  rows = Math.ceil(H / CELL);
  draw();
}

function draw() {
  ctx.clearRect(0, 0, W, H); // fond transparent

  // Hauteur de la bande : petite au repos, grandit pendant le scroll
  const reach = BASE + (BOOST - BASE) * energy;
  const bandH = H * reach;

  // Les cellules sont "accrochées" à la page : quand on scrolle, le motif défile,
  // ce qui donne l'impression que les pixels bougent.
  const scrollRow = Math.floor(scrollY / CELL);

  ctx.fillStyle = PIXEL_COLOR;
  for (let y = 0; y < rows; y++) {
    const dist = H - (y + 1) * CELL; // distance au bord bas de l'écran
    if (dist >= bandH) continue;
    const vis = 1 - Math.max(0, dist) / bandH; // dense en bas, rare en haut
    for (let x = 0; x < cols; x++) {
      if (hash(x, y + scrollRow) > vis) continue;
      ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
    }
  }
}

function loop() {
  target *= 0.9; // la cible retombe quand on arrête de scroller
  energy += (target - energy) * SMOOTH;
  draw();
  if (energy > 0.003 || target > 0.003) {
    requestAnimationFrame(loop);
  } else {
    energy = 0;
    target = 0;
    running = false;
    draw(); // état de repos
  }
}

addEventListener(
  "scroll",
  () => {
    const dy = Math.abs(scrollY - lastY);
    lastY = scrollY;
    target = Math.max(target, Math.min(1, dy / SENSITIVITY));
    if (!running) {
      running = true;
      requestAnimationFrame(loop);
    }
  },
  { passive: true }
);

addEventListener("resize", setup);
setup();
